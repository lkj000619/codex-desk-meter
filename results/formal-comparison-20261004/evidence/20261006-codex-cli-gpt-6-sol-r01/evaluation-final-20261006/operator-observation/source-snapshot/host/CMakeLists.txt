cmake_minimum_required(VERSION 3.16)
project(cdm_receiver_host C)
enable_testing()
if(NOT DEFINED ENV{IDF_PATH})
  message(FATAL_ERROR "IDF_PATH must point to the declared ESP-IDF v5.3.2 SDK")
endif()
file(TO_CMAKE_PATH "$ENV{IDF_PATH}/components/json/cJSON" CJSON_ROOT)
add_executable(receiver_test test_receiver.c ../main/receiver.c "${CJSON_ROOT}/cJSON.c")
target_include_directories(receiver_test PRIVATE ../main "${CJSON_ROOT}")
if(NOT WIN32)
  target_link_libraries(receiver_test PRIVATE m)
endif()
add_test(NAME receiver_state_and_crc COMMAND receiver_test)
add_executable(receiver_cli receiver_cli.c ../main/receiver.c "${CJSON_ROOT}/cJSON.c")
target_include_directories(receiver_cli PRIVATE ../main "${CJSON_ROOT}")
if(NOT WIN32)
  target_link_libraries(receiver_cli PRIVATE m)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME python_frame_to_firmware_receiver COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_pipeline.py" "$<TARGET_FILE:receiver_cli>")
add_executable(feature_test test_feature.c ../main/feature.c)
target_include_directories(feature_test PRIVATE ../main)
add_test(NAME idle_backlight_feature COMMAND feature_test)
